cmake_minimum_required(VERSION 3.22)
project(pocket_runtime C)
set(CMAKE_C_STANDARD 11)
set(BUILD_WASI none)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory(vendor/wasm3/source)
# prepare-toolchain patches the default yield to call pocket_yield.
add_library(pocket STATIC pocket.c vendor/cjson/cJSON.c)
target_include_directories(pocket PUBLIC . vendor/cjson vendor/wasm3/source)
target_link_libraries(pocket PUBLIC m3)
if(MSVC)
  target_compile_options(pocket PRIVATE /experimental:c11atomics)
endif()
if(NOT MSVC)
  target_link_libraries(pocket PUBLIC m)
endif()
if(ANDROID)
  add_library(pocket-jni SHARED jni.c)
  target_link_libraries(pocket-jni PRIVATE pocket log)
else()
  add_executable(pocket-cli cli.c)
  target_link_libraries(pocket-cli PRIVATE pocket)
endif()
